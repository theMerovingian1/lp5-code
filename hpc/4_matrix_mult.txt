#include <iostream>
#include <cuda_runtime.h>
#include <chrono>

using namespace std;

// CPU matrix multiplication
void matMulCPU(int* A, int* B, int* C, int N) {
    for (int row = 0; row < N; ++row) {
        for (int col = 0; col < N; ++col) {
            int sum = 0;
            for (int k = 0; k < N; ++k) {
                sum += A[row * N + k] * B[k * N + col];
            }
            C[row * N + col] = sum;
        }
    }
}

// GPU matrix multiplication
__global__ void matMulGPU(int* A, int* B, int* C, int N) {
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;

    if (row < N && col < N) {
        int sum = 0;
        for (int k = 0; k < N; ++k) {
            sum += A[row * N + k] * B[k * N + col];
        }
        C[row * N + col] = sum;
    }
}

void checkCuda(cudaError_t err, const char* msg) {
    if (err != cudaSuccess) {
        std::cerr << "CUDA error at " << msg << ": " << cudaGetErrorString(err) << std::endl;
        exit(EXIT_FAILURE);
    }
}

int main() {
    const int N = 2;
    int A[N * N] = {3, 6, 7, 5};
    int B[N * N] = {3, 5, 6, 2};
    int C[N * N] = {0};
    int C_cpu[N * N] = {0};

    int *d_A, *d_B, *d_C;
    size_t size = N * N * sizeof(int);

    checkCuda(cudaMalloc(&d_A, size), "cudaMalloc A");
    checkCuda(cudaMalloc(&d_B, size), "cudaMalloc B");
    checkCuda(cudaMalloc(&d_C, size), "cudaMalloc C");

    checkCuda(cudaMemcpy(d_A, A, size, cudaMemcpyHostToDevice), "copy A");
    checkCuda(cudaMemcpy(d_B, B, size, cudaMemcpyHostToDevice), "copy B");

    // Measure CPU time
    auto start_cpu = chrono::high_resolution_clock::now();
    matMulCPU(A, B, C_cpu, N);
    auto end_cpu = chrono::high_resolution_clock::now();
    chrono::duration<double> cpu_duration = end_cpu - start_cpu;

    cout << "Matrix A:\n";
    for (int i = 0; i < N * N; ++i)
        cout << A[i] << (i % N == N - 1 ? "\n" : " ");

    cout << "Matrix B:\n";
    for (int i = 0; i < N * N; ++i)
        cout << B[i] << (i % N == N - 1 ? "\n" : " ");

    // Measure GPU time
    dim3 threads(1, 1);
    dim3 blocks(N, N);
    auto start_gpu = chrono::high_resolution_clock::now();
    matMulGPU<<<blocks, threads>>>(d_A, d_B, d_C, N);
    checkCuda(cudaGetLastError(), "Kernel launch");
    checkCuda(cudaDeviceSynchronize(), "Kernel execution");

    checkCuda(cudaMemcpy(C, d_C, size, cudaMemcpyDeviceToHost), "copy C");

    auto end_gpu = chrono::high_resolution_clock::now();
    chrono::duration<double> gpu_duration = end_gpu - start_gpu;

    cout << "Matrix C (GPU result):\n";
    for (int i = 0; i < N * N; ++i)
        cout << C[i] << (i % N == N - 1 ? "\n" : " ");

    cout << "Matrix C (CPU result):\n";
    for (int i = 0; i < N * N; ++i)
        cout << C_cpu[i] << (i % N == N - 1 ? "\n" : " ");

    // Compare CPU and GPU results
    bool match = true;
    for (int i = 0; i < N * N; ++i) {
        if (C[i] != C_cpu[i]) {
            match = false;
            break;
        }
    }

    if (match) {
        cout << "CPU and GPU results match!" << endl;
    } else {
        cout << "CPU and GPU results do NOT match!" << endl;
    }

    // Print execution times
    cout << "CPU Execution Time: " << cpu_duration.count() << " seconds" << endl;
    cout << "GPU Execution Time: " << gpu_duration.count() << " seconds" << endl;

    // Free memory
    cudaFree(d_A);
    cudaFree(d_B);
    cudaFree(d_C);

    return 0;
}
